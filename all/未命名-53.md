```mermaid
flowchart TD
    Start([开始服务器数据恢复]) --> CheckAuth[检查API认证状态]
    CheckAuth --> AuthValid{认证有效?}
    
    AuthValid -->|否| RequestLogin[请求用户登录]
    AuthValid -->|是| RequestServerSnapshot[请求服务器创建数据快照]
    
    RequestLogin --> UserLogin[用户执行登录]
    UserLogin --> LoginSuccess{登录成功?}
    LoginSuccess -->|否| ShowAuthError[显示认证错误]
    LoginSuccess -->|是| RequestServerSnapshot
    
    RequestServerSnapshot --> SnapshotCreated{快照创建成功?}
    SnapshotCreated -->|否| ShowSnapshotError[显示快照错误]
    SnapshotCreated -->|是| BeginServerTransaction[开始服务器事务]
    
    BeginServerTransaction --> RestoreBasicTables[恢复基本表数据]
    
    subgraph RestoreServerCanvas [恢复服务器canvas表]
        RSC1[准备canvas表解密数据] --> RSC2[解析JSON数据]
        RSC2 --> RSC3[验证数据格式]
        RSC3 --> RSC4{数据有效?}
        RSC4 -->|否| RSC5[记录错误并回滚]
        RSC4 -->|是| RSC6[构建API请求]
        RSC6 --> RSC7[发送清除表请求]
        RSC7 --> RSC8[发送写入数据请求]
        RSC8 --> RSC9{API响应成功?}
        RSC9 -->|否| RSC10[重试或回滚]
        RSC9 -->|是| RSC11[更新恢复进度]
    end
    
    RestoreBasicTables --> RestoreServerCanvas
    
    RestoreServerCanvas --> RestoreServerCanvasComp[恢复服务器canvas_components表]
    RestoreServerCanvasComp --> RestoreServerSidebarComp[恢复服务器sidebar_components表]
    RestoreServerSidebarComp --> RestoreServerHash[恢复服务器hash_tables表]
    
    RestoreServerHash --> RestoreServerCompData[恢复服务器组件数据]
    
    subgraph RestoreServerCompData [恢复服务器组件数据]
        RSCD1[按组件和表分组获取解密数据] --> RSCD2[处理下一个组件表]
        RSCD2 --> RSCD3{还有组件表?}
        RSCD3 -->|否| RSCD13[完成组件数据恢复]
        RSCD3 -->|是| RSCD4[解析组件表数据]
        RSCD4 --> RSCD5[合并多块数据(如有)]
        RSCD5 --> RSCD6[验证合并后数据]
        RSCD6 --> RSCD
```
