function determineBatchSize(totalSize, connectionSpeed) {
  // 基础批次: 1MB 或 500条记录，取较小者
  let baseBatchSize = Math.min(1024 * 1024, 500);
  
  // 根据网络速度调整
  if (connectionSpeed < 1) { // 低于1Mbps
    return Math.floor(baseBatchSize / 2);
  } else if (connectionSpeed > 10) { // 高于10Mbps
    return Math.min(baseBatchSize * 2, 2048 * 1024); // 最大不超过2MB
  }
  
  return baseBatchSize;
}
