```mermaid
flowchart TD
    Start([开始上传流程]) --> CheckLogin[检查登录状态]
    CheckLogin --> LoginStatus{是否已登录?}
    
    LoginStatus -->|否| RequestLogin[请求用户登录]
    LoginStatus -->|是| InitializeProcess[初始化上传流程]
    
    RequestLogin --> UserLogin[用户登录]
    UserLogin --> InitializeProcess
    
    InitializeProcess --> CollectLocalHashes[收集本地组件表哈希]
    CollectLocalHashes --> FetchServerHashes[获取服务器哈希]
    
    FetchServerHashes --> ServerHasData{服务器有数据?}
    
    ServerHasData -->|是| CompareHashes[比较哈希值]
    ServerHasData -->|否| PrepareAllData[准备上传所有数据]
    
    CompareHashes --> ClassifyTables[分类组件表]
    ClassifyTables --> HasChanges{有需同步的表?}
    
    HasChanges -->|否| SkipUpload[跳过上传]
    HasChanges -->|是| PrepareChangedData[准备需同步的数据]
    
    PrepareAllData --> CalculateSize[计算总数据量]
    PrepareChangedData --> CalculateSize
    
    CalculateSize --> EstimateBatchSize[估算最佳批次大小]
    EstimateBatchSize --> InitUploadTracker[初始化上传追踪器]
    InitUploadTracker --> StartBatching[开始批量上传]
    
    subgraph BatchUpload [批量上传流程]
        B1[获取下一批数据] --> B2{还有数据?}
        B2 -->|否| BatchesComplete[所有批次完成]
        B2 -->|是| B3[构建批次包]
        B3 --> B4[上传批次到服务器]
        B4 --> B5{上传成功?}
        B5 -->|是| B6[更新进度]
        B5 -->|否| B7[处理上传失败]
        B7 --> B8{重试上传?}
        B8 -->|是| B4
        B8 -->|否| B9[标记错误继续]
        B6 --> B1
        B9 --> B1
    end
    
    StartBatching --> BatchUpload
    
    BatchUpload --> UpdateServerHashes[更新服务器哈希记录]
    UpdateServerHashes --> VerifyUpload[验证上传完整性]
    
    VerifyUpload --> UploadComplete{上传完成?}
    UploadComplete -->|是| FinalizeMode[完成模式切换]
    UploadComplete -->|部分完成| HandlePartial[处理部分完成]
    
    HandlePartial --> UserChoice{用户选择}
    UserChoice -->
```
