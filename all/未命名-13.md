# ─────────────────────────────────────────────
# 工具函数：从 Message 中提取文本内容
# ─────────────────────────────────────────────
def extract_text(message: Message) -> str:
    for part in message.parts:
        if hasattr(part.root, 'text'):
            return part.root.text
    return ""


def make_text_message(task_id: str, context_id: str, text: str) -> Message:
    """构造一条包含纯文本的 agent Message"""
    return Message(
        role=Role.agent,
        message_id=str(uuid.uuid4()),
        task_id=task_id,
        context_id=context_id,
        parts=[Part(root=TextPart(text=text))],
    )

        http_handler=handler,

# 4. 客户端：向 Agent 发送问题并获取答案
# ─────────────────────────────────────────────
async def ask_agent(url: str, question: str) -> str:
    """向指定 Agent 发送问题，等待任务完成后返回答案"""
    async with httpx.AsyncClient() as http_client:
        client = A2AClient(httpx_client=http_client, url=url)

        # 构造请求
        request = SendMessageRequest(
            id=str(uuid.uuid4()),
            params=MessageSendParams(
                message=Message(
                    role=Role.user,
                    message_id=str(uuid.uuid4()),
                    parts=[Part(root=TextPart(text=question))],
                )
            ),
        )

        response = await client.send_message(request)

        # 从响应中提取文本
        result = response.root
        if hasattr(result, 'result'):
            r = result.result
            # 响应是 Task 或 Message
            if hasattr(r, 'root'):
                r = r.root
            # Task：从 status.message 取文本
            if hasattr(r, 'status') and r.status and r.status.message:
                return extract_text(r.status.message)
            # Message：直接取文本
            if hasattr(r, 'parts'):
                return extract_text(r)

        return f"（无法解析响应：{response}）"


# ─────────────────────────────────────────────
# 5. LLM 路由：自动判断问题应路由到哪个 Agent
# ─────────────────────────────────────────────
AGENTS = {
    "tech": ("http://localhost:7000/", "技术专家"),
    "sales": ("http://localhost:7001/", "销售顾问"),
}


def llm_route(question: str) -> tuple[str, str]:
    """
    用 LLM 判断问题应路由到哪个 Agent。
    返回：(agent_url, agent_name)
    """
    prompt = f"""你是客服系统的路由分类器。
根据客户的问题，判断应路由到哪个专家：
- tech：技术专家（回答 API 调用、集成、安装、配置等技术问题）
- sales：销售顾问（回答价格、套餐、试用、购买、优惠等商务问题）

客户问题：{question}

只需输出一个单词：tech 或 sales（不要包含其他内容）"""

    result = llm.invoke([{"role": "user", "content": prompt}]).strip().lower()
    # 容错处理：如果 LLM 输出不是标准关键词，默认技术专家
    if "sales" in result:
        key = "sales"
    else:
        key = "tech"
    url, name = AGENTS[key]
    return url, name


async def handle_query(question: str):
    url, agent_name = llm_route(question)   # LLM 自动路由
    print(f"\n客户咋询：{question}")
    print(f"→ LLM 路由到：{agent_name}")
    print("=" * 50)
    answer = await ask_agent(url, question)
    print(f"回复：\n{answer}")
    print("=" * 50)

# 6. 启动
# ─────────────────────────────────────────────
def run_server(app, port: int):
    """在独立线程中运行 uvicorn 服务（兼容 PyCharm asyncio 调试模式）"""
    loop = asyncio.new_event_loop()
    asyncio.set_event_loop(loop)
    config = uvicorn.Config(app, host="0.0.0.0", port=port, log_level="error")
    server = uvicorn.Server(config)
    loop.run_until_complete(server.serve())


if __name__ == "__main__":
    # 启动两个 Agent Server（daemon 线程）
    threading.Thread(target=run_server, args=(tech_app, 7000), daemon=True).start()
    threading.Thread(target=run_server, args=(sales_app, 7001), daemon=True).start()

    print("🚀 启动技术专家 Agent → http://localhost:7000")
    print("🚀 启动销售顾问 Agent → http://localhost:7001")
    time.sleep(2)  # 等待 Server 就绪
    print("✅ Agent 就绪，开始测试\n")

    # 测试查询
