// 同步处理函数
async function processSync() {
  // 获取所有待同步的变更记录
  const pendingChanges = await db.query(
    "SELECT * FROM ChangesTracker WHERE sync_status = 'pending' ORDER BY created_at ASC"
  );
  
  for (const change of pendingChanges) {
    // 读取当前组件数据
    const currentComponent = await db.query(
      "SELECT * FROM Components WHERE id = ?", 
      [change.entity_id]
    );
    
    // 计算当前组件的哈希
    const currentHash = computeLayeredHash(currentComponent);
    
    // 情况1: 当前哈希与目标哈希相同 - 已经同步过了或被其他进程同步了
    if (currentHash === change.target_hash) {
      await db.query(
        "UPDATE ChangesTracker SET sync_status = 'synced', sync_time = NOW() WHERE id = ?",
        [change.id]
      );
      continue;
    }
    
    // 情况2: 当前哈希与源哈希相同 - 可以安全同步
    if (currentHash === change.source_hash) {
      // 执行实际同步操作
      await syncComponent(change.entity_id);
      
      // 同步后再次读取组件并计算哈希
      const syncedComponent = await db.query(
        "SELECT * FROM Components WHERE id = ?", 
        [change.entity_id]
      );
      const syncedHash = computeLayeredHash(syncedComponent);
      
      // 更新变更记录状态
      await db.query(
        "UPDATE ChangesTracker SET target_hash = ?, sync_status = 'synced', sync_time = NOW() WHERE id = ?",
        [syncedHash, change.id]
      );
    } 
    // 情况3: 当前哈希与源哈希不同、与目标哈希也不同 - 数据已被修改，存在冲突
    else {
      await db.query(
        "UPDATE ChangesTracker SET sync_status = 'conflict' WHERE id = ?",
        [change.id]
      );
      // 可选：记录日志或通知管理员处理冲突
      logConflict(change.entity_id, change.source_hash, change.target_hash, currentHash);
    }
  }
}
