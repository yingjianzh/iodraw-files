```mermaid
sequenceDiagram
    participant Client as 客户端
    participant AuthService as AuthService
    participant Redis as Redis
    participant MongoDB as MongoDB
    participant CryptoServer as CryptoServer

    Client->>AuthService: POST /login (LoginReq)
    AuthService->>Redis: GET nonce:{address}
    Redis-->>AuthService: nonce
    alt nonce 不匹配
        AuthService-->>Client: 401 NonceMismatch
    else nonce 有效
        AuthService->>Redis: DEL nonce:{address}
        AuthService->>AuthService: VerifySignature()
        alt 签名无效
            AuthService-->>Client: 401 InvalidSignature
        else 签名有效
            AuthService->>MongoDB: FindByWalletAddress()
            alt 用户不存在
                AuthService->>CryptoServer: CreateCaAddress()
                CryptoServer-->>AuthService: caAddress
                AuthService->>MongoDB: 事务(用户注册+初始化)
            else 用户存在
                alt 用户禁用
                    AuthService-->>Client: 403 UserIsDisabled
                else 正常登录
                    AuthService->>AuthService: 生成 access_token/refresh_token
                    AuthService->>Redis: SET refresh:{userID} {refresh_token}
                    AuthService-->>Client: 
                        Note left of Client: HTTP/1.1 200 OK\nSet-Cookie: access_token=xxx; HttpOnly; Secure\nSet-Cookie: refresh_token=yyy; HttpOnly; Secure; Path=/refresh
                end
            end
        end
    end
```
