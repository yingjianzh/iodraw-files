```mermaid
sequenceDiagram
    participant Client as Client
    participant AuthService as AuthService
    participant Redis as Redis
    participant MongoDB as MongoDB
    participant CryptoServer as CryptoServer

    Client->>AuthService: POST /login {address, signature, nonce}
    AuthService->>Redis: GET nonce:{address}
    Redis-->>AuthService: stored_nonce
    
    alt stored_nonce != received_nonce
        AuthService-->>Client: 401 NonceMismatch
    else
        AuthService->>Redis: DEL nonce:{address}
        AuthService->>AuthService: verifySignature(address, signature)
        
        alt signature invalid
            AuthService-->>Client: 401 InvalidSignature
        else
            AuthService->>MongoDB: findUserByAddress(address)
            
            alt user not found
                AuthService->>CryptoServer: createCAAddress(userID)
                CryptoServer-->>AuthService: caAddress
                AuthService->>MongoDB: beginTransaction()
                MongoDB-->>AuthService: transaction_ok
                AuthService->>MongoDB: insertUser(userData)
                AuthService->>MongoDB: initUserPoints()
                AuthService->>MongoDB: commitTransaction()
            else
                alt user.status == disabled
                    AuthService-->>Client: 403 UserDisabled
                else
                    AuthService->>AuthService: generateTokens()
                    AuthService->>Redis: SETEX refresh:{userID} 604800 refresh_token
                    AuthService-->>Client: HTTP/1.1 200 OK
                    Note over Client,AuthService: Set-Cookie: access_token=xxx; HttpOnly; Secure; SameSite=Strict\nSet-Cookie: refresh_token=yyy; HttpOnly; Secure; Path=/refresh; SameSite=Strict
                end
            end
        end
    end
```
