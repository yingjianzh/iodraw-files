**数据结构设计：**
```
type MessageBuilder struct {
    // 基本信息
    ConversationID  string
    MessageID       string
    
    // 内容累加
    Content         strings.Builder  // 累加所有 LLM_CHUNK 的 content
    
    // 结构化数据
    ToolCalls       []ToolCall       // 收集所有 TOOL_CALL
    TaskInfo        *TaskPlan        // TASK_PLAN 事件
    
    // 元数据
    StartTime       time.Time
    EventCount      int              // 已处理的事件数
    StartEventID    int64            // 第一个事件的 ID (仅用于日志)
    
    // 不保存到 DB 的运行时状态
    LastPushedID    int64            // 最后成功推送的 event_id
    IsBlocked       bool             // 是否因推送失败而阻塞
    BlockedAt       int64            // 阻塞在哪个 event_id
}

// Event Manager 维护
type EventManager struct {
    // 会话级消息构建器
    messageBuilders  sync.Map  // conversation_id -> *MessageBuilder
    
    // 会话级事件计数器
    eventCounters    sync.Map  // conversation_id -> *atomic.Int64
    
    // 订阅者
    subscribers      sync.Map  // conversation_id -> chan Event
}
```
